## 流れ

```bash
Step 0  初期設定
Step 1  マップ・ラスタ・ポリゴン確認
Step 2  ポリゴンを1個ずつマスク用FCとして保存
Step 3  マスクごとにComposite rasterを切り抜き
Step 4  切り抜きラスタからB/G/R/REDEDGE/NIRを抽出
Step 5  BバンドをRaster to Point
Step 6  ポイントにB/G/R/REDEDGE/NIRの値を追加
```

## Step 1  マップ・ラスタ・ポリゴン確認

```bash
# 関数

def safe_name(name, max_len=80):
    name = os.path.splitext(os.path.basename(str(name)))[0]
    name = re.sub(r"[^\w\u3040-\u30ff\u3400-\u9fff]", "_", name)
    name = re.sub(r"_+", "_", name)
    name = name.strip("_")

    if not name:
        name = "unnamed"

    if re.match(r"^\d", name):
        name = "n_" + name

    return name[:max_len]


def unique_path(workspace, base_name):
    base_name = safe_name(base_name)
    out_path = os.path.join(workspace, base_name)

    if not arcpy.Exists(out_path):
        return out_path

    i = 1
    while True:
        candidate = os.path.join(workspace, f"{base_name}_{i}")
        if not arcpy.Exists(candidate):
            return candidate
        i += 1


# マップ取得

aprx = arcpy.mp.ArcGISProject("CURRENT")

maps = aprx.listMaps(map_name)

if not maps:
    print("現在のプロジェクト内のマップ一覧:")
    for mm in aprx.listMaps():
        print(f"  - {mm.name}")
    raise ValueError(f"マップ '{map_name}' が見つかりません。")

m = maps[0]

print(f"対象マップ: {m.name}")


# ラスタ取得

composite_raster_path = None

print("\n=== ラスタレイヤー一覧 ===")

for lyr in m.listLayers():
    if lyr.isRasterLayer:
        desc = arcpy.Describe(lyr)
        print("-----------------------------")
        print("Layer name :", lyr.name)
        print("CatalogPath:", desc.catalogPath)

        if lyr.name == composite_raster_name:
            composite_raster_path = desc.catalogPath

if composite_raster_path is None:
    raise ValueError(f"元ラスタ '{composite_raster_name}' が見つかりません。")

print("\n元マルチバンドラスタ:")
print(composite_raster_path)


# ポリゴンレイヤー取得

polygon_layers = []

print("\n=== ポリゴンレイヤー一覧 ===")

for lyr in m.listLayers():
    if lyr.isFeatureLayer:
        desc = arcpy.Describe(lyr)
        if hasattr(desc, "shapeType") and desc.shapeType == "Polygon":
            polygon_layers.append(lyr)
            print(f"  - {lyr.name}")

if not polygon_layers:
    raise ValueError("ポリゴンレイヤーが見つかりません。")

print(f"\nポリゴンレイヤー数: {len(polygon_layers)}")
```

## Step 2  ポリゴンを1個ずつマスク用FCとして保存

```bash
# Step 2. 各ポリゴンを個別マスクとして保存

mask_fcs = []

for poly_lyr in polygon_layers:

    poly_desc = arcpy.Describe(poly_lyr)
    poly_path = poly_desc.catalogPath
    oid_field = poly_desc.OIDFieldName
    poly_layer_name = safe_name(poly_lyr.name)

    print(f"\n処理中のポリゴンレイヤー: {poly_lyr.name}")

    with arcpy.da.SearchCursor(poly_lyr, [oid_field]) as cursor:
        for row in cursor:
            oid = row[0]

            mask_name = f"mask_{poly_layer_name}_{oid}"
            mask_fc = os.path.join(output_gdb, safe_name(mask_name))

            if arcpy.Exists(mask_fc):
                print(f"  既存のためスキップ: {os.path.basename(mask_fc)}")
                mask_fcs.append(mask_fc)
                continue

            where = f"{arcpy.AddFieldDelimiters(poly_path, oid_field)} = {oid}"

            temp_layer = arcpy.management.MakeFeatureLayer(
                poly_lyr,
                f"temp_{mask_name}",
                where
            )[0]

            arcpy.management.CopyFeatures(temp_layer, mask_fc)

            mask_fcs.append(mask_fc)

            print(f"  作成: {os.path.basename(mask_fc)}")

print("\nマスク作成完了")
print(f"マスク数: {len(mask_fcs)}")
```

## Step 3  マスクごとにComposite rasterを切り抜き

```bash
# Step 3. Composite rasterをマスクで切り抜き

clipped_composites = []

for mask_fc in mask_fcs:

    mask_base = safe_name(os.path.basename(mask_fc))
    out_name = f"{mask_base}_Composite_clip"
    out_raster = os.path.join(output_gdb, safe_name(out_name))

    if arcpy.Exists(out_raster):
        print(f"既存のためスキップ: {os.path.basename(out_raster)}")
        clipped_composites.append(out_raster)
        continue

    print(f"\nExtractByMask 開始: {mask_base}")

    clipped = ExtractByMask(composite_raster_path, mask_fc)
    clipped.save(out_raster)

    clipped_composites.append(out_raster)

    print(f"完了: {os.path.basename(out_raster)}")

print("\nComposite raster の切り抜き完了")
print(f"出力数: {len(clipped_composites)}")
```

## Step 4  切り抜きラスタからB/G/R/REDEDGE/NIRを抽出

```bash
# Step 4. 各Composite clipからB/G/R/REDEDGE/NIRを抽出

band_raster_dict = {}

for clipped_composite in clipped_composites:

    comp_base = safe_name(os.path.basename(clipped_composite))
    mask_base = comp_base.replace("_Composite_clip", "")

    band_raster_dict[mask_base] = {}

    print(f"\nバンド抽出開始: {mask_base}")

    for band_number, band_name in band_settings:

        out_name = f"{mask_base}_{band_name}_clip"
        out_band = os.path.join(
            output_gdb,
            safe_name(out_name)
        )

        # すでに存在する場合
        if arcpy.Exists(out_band):

            print(
                f"  既存のためスキップ: "
                f"{os.path.basename(out_band)}"
            )

            band_raster_dict[mask_base][band_name] = out_band
            continue

        print(
            f"  ExtractBand: "
            f"Band {band_number} -> {band_name}"
        )

        try:
            # 第1引数は in_raster ではなく raster
            band_raster = arcpy.ia.ExtractBand(
                raster=clipped_composite,
                band_ids=[band_number]
            )

            band_raster.save(out_band)

            band_raster_dict[mask_base][band_name] = out_band

            print(
                f"  完了: "
                f"{os.path.basename(out_band)}"
            )

        except Exception as e:

            print(
                f"  抽出失敗: "
                f"{mask_base} / {band_name}"
            )

            print(f"  エラー: {e}")

            raise


print("\nバンド抽出完了")
```

## Step 5  1つのバンドをRaster to Point

```bash
# Step 5. BバンドをRaster to Point

point_fcs = []

for mask_base, bands in band_raster_dict.items():

    b_raster = bands["B"]

    out_name = f"{mask_base}_points"
    out_points = os.path.join(output_gdb, safe_name(out_name))

    if arcpy.Exists(out_points):
        print(f"既存のためスキップ: {os.path.basename(out_points)}")
        point_fcs.append(out_points)
        continue

    print(f"\nRasterToPoint 開始: {mask_base}")

    arcpy.conversion.RasterToPoint(
        in_raster=b_raster,
        out_point_features=out_points,
        raster_field="VALUE"
    )

    point_fcs.append(out_points)

    print(f"完了: {os.path.basename(out_points)}")

print("\nRaster to Point 完了")
print(f"ポイントFC数: {len(point_fcs)}")
```

## Step 6  ポイントにB/G/R/REDEDGE/NIRの値を追加

```bash
# Step 6. Extract Multi Values to Points

final_point_outputs = []

for point_fc in point_fcs:

    point_base = safe_name(os.path.basename(point_fc))
    mask_base = point_base.replace("_points", "")

    if mask_base not in band_raster_dict:
        print(f"対応するバンドラスタが見つからないためスキップ: {point_base}")
        continue

    bands = band_raster_dict[mask_base]

    print(f"\nExtractMultiValuesToPoints 開始: {point_base}")

    raster_value_list = [
        [bands["B"], "B"],
        [bands["G"], "G"],
        [bands["R"], "R"],
        [bands["REDEDGE"], "REDEDGE"],
        [bands["NIR"], "NIR"],
    ]

    ExtractMultiValuesToPoints(
        in_point_features=point_fc,
        in_rasters=raster_value_list,
        bilinear_interpolate_values="NONE"
    )

    final_point_outputs.append(point_fc)

    print(f"完了: {os.path.basename(point_fc)}")

print("\nすべて完了")
print("最終ポイント:")

for fc in final_point_outputs:
    print(fc)
```

## Step 7  csvで保存

```bash
# -*- coding: utf-8 -*-

import arcpy
import os
import re


# ============================================================
# 0. 設定
# ============================================================

# 対象マップ名
map_name = "SiteE"

# 入力する名前
# 例: "a_1", "b_3", "e_1"
input_name = "e_1"

# 現在のArcGIS Proプロジェクトを取得
aprx = arcpy.mp.ArcGISProject("CURRENT")

# 出力先GDB
# 現在のプロジェクトのデフォルトGDBを使用
output_gdb = aprx.defaultGeodatabase

# 座標系
# Noneの場合は対象マップの座標系を使用
spatial_reference = None

# 作成したフィーチャクラスをマップに追加するか
add_to_map = True


# ============================================================
# 1. 関数
# ============================================================

def safe_name(name):
    """
    ArcGISで使用可能な名前に整える。
    """

    # 英数字・アンダースコア以外を「_」に置換
    name = re.sub(r"[^\w]", "_", str(name))

    # 連続するアンダースコアを1つにする
    name = re.sub(r"_+", "_", name)

    # 先頭と末尾のアンダースコアを削除
    name = name.strip("_")

    # 数字から始まる場合は先頭にアンダースコアを追加
    if re.match(r"^\d", name):
        name = "_" + name

    return name


def normalize_path(path):
    """
    パスを比較しやすい形に統一する。
    """

    return os.path.normcase(
        os.path.normpath(path)
    )


def get_existing_layer_sources(target_map):
    """
    マップ内にすでに存在するレイヤーの
    データソース一覧を取得する。
    """

    existing_sources = set()

    for layer in target_map.listLayers():

        if not layer.isFeatureLayer:
            continue

        try:
            source = normalize_path(layer.dataSource)
            existing_sources.add(source)

        except Exception:
            # グループレイヤーや壊れたリンクなどは無視
            pass

    return existing_sources


# ============================================================
# 2. マップ取得
# ============================================================

maps = aprx.listMaps(map_name)

if not maps:

    print("現在のプロジェクト内のマップ一覧:")

    for project_map in aprx.listMaps():
        print(f"  - {project_map.name}")

    raise ValueError(
        f"マップ '{map_name}' が見つかりません。"
    )

m = maps[0]

# 座標系が指定されていない場合は、マップの座標系を使用
if spatial_reference is None:
    spatial_reference = m.spatialReference


print("=" * 60)
print("処理設定")
print("=" * 60)
print(f"対象マップ: {m.name}")
print(f"出力GDB: {output_gdb}")
print(f"座標系: {spatial_reference.name}")


# ============================================================
# 3. 入力名を分解
# ============================================================

# 前後の空白を削除
input_name = input_name.strip()

# 例: e_1 → alphabet = e、number = 1
parts = input_name.split("_")

if len(parts) != 2:
    raise ValueError(
        "input_nameは 'e_1' のような、"
        "文字_番号の形式で入力してください。"
    )

alphabet = safe_name(parts[0])
number = safe_name(parts[1])

if not alphabet or not number:
    raise ValueError(
        "input_nameの文字または番号が正しくありません。"
    )


print("\n入力:")
print(f"  文字: {alphabet}")
print(f"  番号: {number}")


# ============================================================
# 4. 作成するポリゴン名リスト
# ============================================================

polygon_names = []

# e_1を入力した場合:
#
# e_1_1_1
# e_1_1_2
# e_1_1_3
# e_1_1_4
# e_1_8_1
# e_1_8_2
# e_1_8_3
# e_1_8_4

for main_num in [1, 8]:

    for sub_num in [1, 2, 3, 4]:

        polygon_name = (
            f"{alphabet}_{number}_{main_num}_{sub_num}"
        )

        polygon_names.append(
            safe_name(polygon_name)
        )


print("\n作成対象のポリゴンフィーチャクラス:")

for polygon_name in polygon_names:
    print(f"  - {polygon_name}")


# ============================================================
# 5. ポリゴンフィーチャクラスを作成
# ============================================================

# 作成対象すべて
all_fcs = []

# 今回新しく作成されたものだけ
newly_created_fcs = []

# すでに存在していたもの
existing_fcs = []


print("\n" + "=" * 60)
print("フィーチャクラス作成")
print("=" * 60)

for polygon_name in polygon_names:

    out_fc = os.path.join(
        output_gdb,
        polygon_name
    )

    all_fcs.append(out_fc)

    # すでにGDB内に存在する場合
    if arcpy.Exists(out_fc):

        print(
            f"既存のため作成をスキップ: "
            f"{polygon_name}"
        )

        existing_fcs.append(out_fc)
        continue

    print(f"作成中: {polygon_name}")

    try:
        arcpy.management.CreateFeatureclass(
            out_path=output_gdb,
            out_name=polygon_name,
            geometry_type="POLYGON",
            spatial_reference=spatial_reference
        )

        newly_created_fcs.append(out_fc)

        print(f"  作成完了: {out_fc}")

    except Exception as e:

        print(f"  作成失敗: {polygon_name}")
        print(f"  エラー: {e}")

        raise


# ============================================================
# 6. マップに追加
# ============================================================

if add_to_map:

    print("\n" + "=" * 60)
    print("マップへの追加")
    print("=" * 60)

    # 現在マップに存在するレイヤーのデータソースを取得
    existing_layer_sources = get_existing_layer_sources(m)

    for fc in all_fcs:

        normalized_fc = normalize_path(fc)
        layer_name = os.path.basename(fc)

        # 同じフィーチャクラスがすでにマップにある場合
        if normalized_fc in existing_layer_sources:

            print(
                f"すでにマップ内にあるためスキップ: "
                f"{layer_name}"
            )

            continue

        try:
            m.addDataFromPath(fc)

            # 追加済み一覧に登録
            existing_layer_sources.add(normalized_fc)

            print(f"マップに追加: {layer_name}")

        except Exception as e:

            print(f"マップに追加できませんでした: {fc}")
            print(f"エラー: {e}")


# ============================================================
# 7. プロジェクトを保存
# ============================================================

try:
    aprx.save()
    print("\nArcGIS Proプロジェクトを保存しました。")

except Exception as e:
    print("\nプロジェクトを保存できませんでした。")
    print(f"エラー: {e}")


# ============================================================
# 8. 結果表示
# ============================================================

print("\n" + "=" * 60)
print("処理結果")
print("=" * 60)

print(f"作成対象数: {len(all_fcs)}")
print(f"新規作成数: {len(newly_created_fcs)}")
print(f"既存データ数: {len(existing_fcs)}")


if newly_created_fcs:

    print("\n今回新しく作成したフィーチャクラス:")

    for fc in newly_created_fcs:
        print(f"  - {fc}")


if existing_fcs:

    print("\nすでに存在していたフィーチャクラス:")

    for fc in existing_fcs:
        print(f"  - {fc}")


print("\nすべての対象フィーチャクラス:")

for fc in all_fcs:
    print(f"  - {fc}")


print("\nすべての処理が完了しました。")
```
